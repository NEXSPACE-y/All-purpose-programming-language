### dynamic namespaced constant
<?php namespace A; define('A\VALUE', 7); echo VALUE;
### dynamic constant over global constant
<?php namespace { const VALUE = 'global'; } namespace A { define('A\VALUE', 'local'); echo VALUE; }
### defined null constant does not fall back
<?php namespace { const VALUE = 'global'; } namespace A { define('A\VALUE', null); var_dump(VALUE); }
### constant fallback remains case sensitive
<?php namespace { const value = 'global'; } namespace A { define('A\VALUE', 'local'); echo VALUE, '|', value; }
### dynamic constant in function body
<?php namespace A; function read(){return VALUE;} define('A\VALUE', 19); echo read();
### repeated constant expression after local definition
<?php namespace { const VALUE = 'global'; } namespace A { function read(){return VALUE;} echo read(), '|'; define('A\VALUE', 'local'); echo read(); }
### conditional local function not installed
<?php namespace A; if (false) { function strlen($s){return 9;} } echo strlen('abc');
### conditional local function installed
<?php namespace A; if (true) { function strlen($s){return 9;} } echo strlen('abc');
### two call sites bind independently
<?php namespace A; echo strlen('abc'), '|'; if (true) { function strlen($s){return 9;} } echo strlen('abc');
### same call site after local function definition
<?php namespace A; function read(){return strlen('abc');} echo read(), '|'; if (true) { function strlen($s){return 9;} } echo read();
### nested declaration resolves at runtime
<?php namespace A; function install(){function inner(){return 13;}} install(); echo inner();
### function fallback into global user function
<?php namespace { function read(){return 17;} } namespace A { if (false) { function read(){return 3;} } echo read(); }
### local reference return selected at runtime
<?php namespace A; $v=1; if (true) { function &read(){global $v; return $v;} } $ref =& read(); $ref=8; echo $v;
### global reference return fallback
<?php namespace { $v=1; function &read(){global $v; return $v;} } namespace A { if (false) { function read(){} } $ref =& read(); $ref=8; echo $v; }
### first class global fallback with unexecuted local declaration
<?php namespace A; if (false) { function strlen($s){return 9;} } $f=strlen(...); echo $f('abc');
### first class local function binds when created
<?php namespace A; if (true) { function strlen($s){return 9;} } $f=strlen(...); echo $f('abc');
### imported constant is resolved exactly
<?php namespace X { const VALUE='import'; } namespace A { use const X\VALUE; define('A\VALUE','local'); echo VALUE; }
### imported function is resolved exactly
<?php namespace X { function read(){return 'import';} } namespace A { use function X\read; if (true) { function own(){return 'local';} } echo read(), '|', own(); }
### qualified constant has no global fallback
<?php namespace A; define('A\Sub\VALUE', 'qualified'); echo Sub\VALUE;
### relative function stays in current namespace
<?php namespace A; if (true) { function read(){return 23;} } echo namespace\read();
### failed call lookup is retried after local declaration
<?php namespace A; function probe($n){try{return late($n);}finally{if($n==0){return 'first';}}} echo probe(0),'|'; if(true){function late($n){return 'local';}} echo probe(1);
### failed first class lookup is retried after local declaration
<?php namespace A; function factory($n){try{return late(...);}finally{if($n==0){return null;}}} factory(0); if(true){function late(){return 'local';}} $f=factory(1); echo $f();
### first class unknown function rejected when created
<?php namespace A; try {$f=missing(...); echo 'created';} catch(\Error $e){echo 'caught';}
### undefined call rejected before argument side effects
<?php namespace A; function arg(){echo 'arg|';return 1;} try {missing(arg());} catch(\Error $e){echo 'caught|';} finally {echo 'finally';}
### successful builtin binding survives argument exception
<?php namespace A; function probe($n){return strlen(1/$n);} try{probe(0);}catch(\DivisionByZeroError $e){echo 'caught|';} if(true){function strlen($s){return 9;}} echo probe(1);
